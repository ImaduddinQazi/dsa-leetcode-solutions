Time Complexity: O(n)
Space Complexity: O(n)

Code:
class Solution {
    public List<Integer> findDuplicates(int[] nums) {
        HashMap<Integer, Integer> hm=new HashMap<>();

        List<Integer> list=new ArrayList<>();
        for(int i=0;i<nums.length;i++){
            if(!hm.containsKey(nums[i])){
                hm.put(nums[i],1);
            }
            else{
                list.add(nums[i]);
            }
        }
        
        return list;
    }
}
