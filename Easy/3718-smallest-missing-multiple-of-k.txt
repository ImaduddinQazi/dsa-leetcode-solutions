Time Complexity: O(n)
Space Complexity: O(n)

Code:
class Solution {
    public int missingMultiple(int[] nums, int k) {
        HashSet<Integer> hs=new HashSet<>();

        for(int i=0;i<nums.length;i++){
            if(nums[i]%k==0) hs.add(nums[i]);
        }
        int x=k;
        while(true){
            if(!hs.contains(x) && x%k==0) break;
            x+=k;
        }
        return x;
    }
}
