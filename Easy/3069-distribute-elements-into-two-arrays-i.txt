Time Complexity: O(n)
Space Complexity: O(n)

Code:
class Solution {
    public int[] resultArray(int[] nums) {
        if(nums.length<=2) return nums;
        Stack<Integer> arr1=new Stack<>();
        Stack<Integer> arr2=new Stack<>();

        arr1.push(nums[0]);
        arr2.push(nums[1]);
        for(int i=2;i<nums.length;i++){
            if(arr1.peek()>arr2.peek()){
                arr1.push(nums[i]);
            }
            else{
                arr2.push(nums[i]);
            }
        }
        int x=nums.length-1;
        while(!arr2.isEmpty()){
            nums[x]=arr2.pop();
            x--;
        }
        while(!arr1.isEmpty()){
            nums[x]=arr1.pop();
            x--;
        }
        return nums;
    }
}
