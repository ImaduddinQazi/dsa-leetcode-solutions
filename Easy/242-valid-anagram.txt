Time Complexity: O(n)
Space Complexity: O(1)

Code:
class Solution {
    public boolean isAnagram(String s, String t) {
        int n=s.length();
        int m=t.length();

        if(n!=m) return false;

        int[] arr=new int[256];

        for(int i=0;i<n;i++){
            arr[s.charAt(i)]++;
            arr[t.charAt(i)]--;
        }
        for(int i=0;i<256;i++){
            if(arr[i]!=0) return false;
        }
        return true;
    }
}
