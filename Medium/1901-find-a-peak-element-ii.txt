Time Complexity: O(n * log (m))
Space Complexity: O(1)

Code:
class Solution {
    public int[] findPeakGrid(int[][] mat) {
        int n = mat.length;
        int m = mat[0].length;

        int low = 0;
        int high = n - 1;

        while (low <= high) {
            int mid = (low + high) / 2;

            int max = findMax(mat[mid]);

            int up = -1;
            if (mid - 1 >= 0) {
                up = mat[mid - 1][max];
            }
            int down = -1;
            if (mid + 1 < n) {
                down = mat[mid + 1][max];
            }

            if (mat[mid][max] > up && mat[mid][max] > down) {
                return new int[] { mid, max };
            } else if (mat[mid][max] < up) {
                high = mid - 1;
            } else {
                low = mid + 1;
            }
        }
        return new int[] { -1, -1 };

    }

    public static int findMax(int[] arr) {
        int max = 0;
        for (int i = 0; i < arr.length; i++) {
            if (arr[i] > arr[max])
                max = i;
        }
        return max;
    }
}
