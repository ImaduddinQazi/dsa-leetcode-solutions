Time Complexity: O(log(m * n))
Space Complexity: O(1)

Code:
class Solution {
    public boolean searchMatrix(int[][] matrix, int target) {
        int n=matrix.length;
        int m=matrix[0].length;
        int low=0;
        int high=(m*n)-1;

        int col=m;
        while(low<=high){
            int mid=(low+high)/2;

            int x=mid/col;
            int y=mid%col;

            if(matrix[x][y]==target){
                return true;
            }
            else if(matrix[x][y]<target){
                low=mid+1;
            }
            else{
                high=mid-1;
            }
        }
        return false;
    }
}
