import java.util.HashSet;
public class Solution {
    public ListNode detectCycle(ListNode head) {
        HashSet<ListNode> hs=new HashSet<>();
        if(head==null || head.next==null) return null;

        ListNode temp=head;
        while(temp!=null){
            if(hs.contains(temp)){
                return temp;
            }
            else{
                hs.add(temp);
            }
            temp=temp.next;
        }
        return null;
    }
}
